{%- comment -%} JSON-LD `Event` schema per card. Powers Google's Event rich result (date / venue / time directly in the SERP). Required fields: name, startDate, location. Time-of-day: `event.time` is free-form (e.g. "21:30", "09:30–13:00", "Evening"). When it parses cleanly as HH:MM we append it to startDate as a local-time ISO string (no TZ offset — Atlantic/Azores observes DST and Google interprets bare ISO times as event-local, which is correct for Terceira). Anything else falls back to date-only. Address: parsed into a `PostalAddress` when a Portuguese 9700-/9760- postal code is present; otherwise the raw string becomes the `addressLocality`. `addressCountry` is always "PT". {%- endcomment -%} {%- comment -%} -- Parse event.time into HH:MM:SS if possible. -- {%- endcomment -%} {%- assign time_iso = "" -%} {%- if event.time and event_kind != "exhibition" -%} {%- assign t_raw = event.time | strip -%} {%- assign t_parts = t_raw | split: ":" -%} {%- if t_parts.size >= 2 -%} {%- assign hh = t_parts[0] | strip -%} {%- assign mm_raw = t_parts[1] | slice: 0, 2 -%} {%- assign digits = "0123456789" -%} {%- assign hh_ok = true -%} {%- if hh.size == 0 or hh.size > 2 -%}{%- assign hh_ok = false -%}{%- endif -%} {%- if hh_ok -%} {%- for i in (0..1) -%} {%- if i < hh.size -%} {%- assign ch = hh | slice: i, 1 -%} {%- unless digits contains ch -%}{%- assign hh_ok = false -%}{%- endunless -%} {%- endif -%} {%- endfor -%} {%- endif -%} {%- assign mm_ok = true -%} {%- if mm_raw.size != 2 -%}{%- assign mm_ok = false -%}{%- endif -%} {%- if mm_ok -%} {%- for i in (0..1) -%} {%- assign ch = mm_raw | slice: i, 1 -%} {%- unless digits contains ch -%}{%- assign mm_ok = false -%}{%- endunless -%} {%- endfor -%} {%- endif -%} {%- if hh_ok and mm_ok -%} {%- assign hh_n = hh | plus: 0 -%} {%- assign mm_n = mm_raw | plus: 0 -%} {%- if hh_n >= 0 and hh_n <= 23 and mm_n >= 0 and mm_n <= 59 -%} {%- assign hh_pad = hh -%} {%- if hh.size == 1 -%}{%- assign hh_pad = "0" | append: hh -%}{%- endif -%} {%- assign time_iso = hh_pad | append: ":" | append: mm_raw | append: ":00" -%} {%- endif -%} {%- endif -%} {%- endif -%} {%- endif -%}
